# 5. 메인 레이아웃 (지도 + 현황 차트)
# ---------------------------------------------------------
col_left, col_right = st.columns([1.2, 1])

# 필터링 적용
filtered_df = region_df[region_df["위험등급"].isin(selected_risk)]
if selected_region != "전체 보기":
    filtered_df = filtered_df[filtered_df["지역"] == selected_region]

with col_left:
    st.subheader("🗺️ 전국 노후 신재생에너지 위험도 GIS 맵")
    
    # Scatter mapbox 생성
    fig_map = px.scatter_mapbox(
        filtered_df,
        lat="lat",
        lon="lon",
        size="노후설비수(15년이상)",
        color="노후화율(%)",
        color_continuous_scale="Reds",
        hover_name="지역",
        hover_data=["주요발전원", "총설비수", "위험등급"],
        zoom=6.2,
        center={"lat": 35.8, "lon": 127.8},
        mapbox_style="carto-positron",
        height=450
    )
    fig_map.update_layout(margin={"r":0,"t":0,"l":0,"b":0})
    st.plotly_chart(fig_map, use_container_width=True)

with col_right:
    st.subheader("📊 지역별 노후화율 및 주요 위험 원인")
    
    # 막대 그래프
    fig_bar = px.bar(
        filtered_df.sort_values(by="노후화율(%)", ascending=True),
        x="노후화율(%)",
        y="지역",
        orientation='h',
        color="주요발전원",
        text="노후화율(%)",
        color_discrete_sequence=px.colors.qualitative.Set2,
        height=450
    )
    fig_bar.update_traces(texttemplate='%{text}%', textposition='outside')
    fig_bar.update_layout(xaxis_range=[0, 110], margin={"r":10,"t":10,"l":10,"b":10})
    st.plotly_chart(fig_bar, use_container_width=True)

st.markdown("---")
